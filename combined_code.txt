def calculate_unit_price(name: str, packsize: str, price: float) -> Dict[str, Any]:
    """
    Use regex to detect weights and volumes (kg, g, gm, l, ltr, ml).
    Convert kg/L to 1000g/1000ml baseline, and compute normalized_price = round((price / base_units) * 100, 2)
    per 100g or 100ml. If no unit matches, return price with unit "per pack".
    """
    if price is None or price <= 0:
        return {"normalized_price": 0.0, "unit": "per pack", "formatted": "â‚¹0.00 / pack"}

    combined = f"{name} {packsize}".lower()

    # Multiplier pattern: e.g. "4 x 70 g", "2 x 500 ml", "2 x 200 ml"
    mult_match = re.search(
        r'(\d+)\s*[xX*]\s*(\d+(?:\.\d+)?)\s*(kg|kilo|kilogram|g|gm|gms|gram|l|ltr|litre|liter|ml)\b',
        combined
    )
    if mult_match:
        multiplier = float(mult_match.group(1))
        qty = float(mult_match.group(2))
        u = mult_match.group(3)
        if u in ['kg', 'kilo', 'kilogram']:
            base_units = multiplier * qty * 1000.0
            norm = round((price / base_units) * 100, 2)
            return {"normalized_price": norm, "unit": "100g", "formatted": f"â‚¹{norm:.2f} / 100g"}
        elif u in ['g', 'gm', 'gms', 'gram']:
            base_units = multiplier * qty
            norm = round((price / base_units) * 100, 2)
            return {"normalized_price": norm, "unit": "100g", "formatted": f"â‚¹{norm:.2f} / 100g"}
        elif u in ['l', 'ltr', 'litre', 'liter']:
            base_units = multiplier * qty * 1000.0
            norm = round((price / base_units) * 100, 2)
            return {"normalized_price": norm, "unit": "100ml", "formatted": f"â‚¹{norm:.2f} / 100ml"}
        elif u == 'ml':
            base_units = multiplier * qty
            norm = round((price / base_units) * 100, 2)
            return {"normalized_price": norm, "unit": "100ml", "formatted": f"â‚¹{norm:.2f} / 100ml"}

    # Kilograms
    kg_match = re.search(r'(\d+(?:\.\d+)?)\s*(?:kg|kilo|kilogram)s?\b', combined)
    if kg_match:
        kg_val = float(kg_match.group(1))
        base_units = kg_val * 1000.0
        if base_units > 0:
            norm = round((price / base_units) * 100, 2)
            return {"normalized_price": norm, "unit": "100g", "formatted": f"â‚¹{norm:.2f} / 100g"}

    # Liters
    l_match = re.search(r'(\d+(?:\.\d+)?)\s*(?:l|ltr|litre|liter)s?\b', combined)
    if l_match:
        l_val = float(l_match.group(1))
        base_units = l_val * 1000.0
        if base_units > 0:
            norm = round((price / base_units) * 100, 2)
            return {"normalized_price": norm, "unit": "100ml", "formatted": f"â‚¹{norm:.2f} / 100ml"}

    # Milliliters
    ml_match = re.search(r'(\d+(?:\.\d+)?)\s*ml\b', combined)
    if ml_match:
        base_units = float(ml_match.group(1))
        if base_units > 0:
            norm = round((price / base_units) * 100, 2)
            return {"normalized_price": norm, "unit": "100ml", "formatted": f"â‚¹{norm:.2f} / 100ml"}

    # Grams
    g_match = re.search(r'(\d+(?:\.\d+)?)\s*(?:g|gm|gms|gram)s?\b', combined)
    if g_match:
        base_units = float(g_match.group(1))
        if base_units > 0:
            norm = round((price / base_units) * 100, 2)
            return {"normalized_price": norm, "unit": "100g", "formatted": f"â‚¹{norm:.2f} / 100g"}

    # Default fallback per pack
    return {
        "normalized_price": round(price, 2),
        "unit": "per pack",
        "formatted": f"â‚¹{price:.2f} / pack"
    }


def _get_blinkit_auth_key() -> str:
    """
    Obtains a temporary anonymous auth_key from Blinkit accounts API to authorize search requests.
    """
    try:
        r = requests.post(
            "https://blinkit.com/v2/accounts/auth_key/",
            headers={
                "app_client": "consumer_web",
                "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0.0.0 Safari/537.36"
            },
            impersonate="chrome124",
            timeout=5
        )
        if r.status_code == 200:
            return r.json().get("auth_key", "")
    except Exception as e:
        logger.warning(f"Failed to obtain Blinkit auth_key: {e}")
    return ""


def fetch_blinkit_sync(query: str, lat: str, lon: str) -> List[Dict[str, Any]]:
    """
    Fetches real live search results from Blinkit layout search API using curl_cffi chrome124 impersonation.
    Defensively traverses response snippets to locate product cards and items.
    """
    encoded_q = urllib.parse.quote(query)
    auth_key = _get_blinkit_auth_key()

    search_headers = {
        "auth_key": auth_key,
        "app_client": "consumer_web",
        "platform": "desktop_web",
        "lat": lat,
        "lon": lon,
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0.0.0 Safari/537.36",
        "Accept": "application/json, text/plain, */*",
        "Content-Type": "application/json",
        "Referer": f"https://blinkit.com/s/?q={encoded_q}",
        "Origin": "https://blinkit.com"
    }
    body = {
        "applied_filters": None,
        "sort": "",
        "postback_meta": {},
        "previous_search_query": "",
        "vertical_cards_processed": 0
    }

    url = f"https://blinkit.com/v1/layout/search?q={encoded_q}&search_type=type_to_search"
    products = []

    try:
        r = requests.post(url, headers=search_headers, json=body, impersonate="chrome124", timeout=8)

        # If Nagpur local coordinate is not serviceable by dark stores, fallback to active geofence
        if r.status_code == 400 and "location not serviceable" in r.text:
            logger.info(f"Blinkit location ({lat}, {lon}) outside active store geofence. Using active geofence ({ACTIVE_GEOFENCE_LAT}, {ACTIVE_GEOFENCE_LON}).")
            search_headers["lat"] = ACTIVE_GEOFENCE_LAT
            search_headers["lon"] = ACTIVE_GEOFENCE_LON
            r = requests.post(url, headers=search_headers, json=body, impersonate="chrome124", timeout=8)

        if r.status_code == 200:
            data = r.json()
            snippets = data.get("response", {}).get("snippets", [])
            for s in snippets:
                sd = s.get("data", {})
                cart_item = sd.get("atc_action", {}).get("add_to_cart", {}).get("cart_item", {})
                name = cart_item.get("product_name") or cart_item.get("display_name")
                price = cart_item.get("price") or cart_item.get("mrp")
                unit = cart_item.get("unit")

                if not name and isinstance(sd.get("name"), dict):
                    name = sd["name"].get("text")
                elif not name and isinstance(sd.get("name"), str):
                    name = sd["name"]

                if not price and isinstance(sd.get("normal_price"), dict):
                    ptxt = sd["normal_price"].get("text", "")
                    num = re.sub(r'[^\d.]', '', ptxt)
                    if num:
                        price = float(num)
                elif not price and isinstance(sd.get("normal_price"), (int, float)):
                    price = float(sd["normal_price"])

                if not unit and isinstance(sd.get("variant"), dict):
                    unit = sd["variant"].get("text")
                elif not unit and isinstance(sd.get("variant"), str):
                    unit = sd["variant"]

                img = (sd.get("image") or {}).get("url") or ""

                if name and price:
                    try:
                        p_val = float(price)
                        if p_val > 5000:
                            p_val /= 100.0
                        products.append({
                            "name": name,
                            "price": round(p_val, 2),
                            "unit": str(unit or ""),
                            "image_url": img,
                            "store": "Blinkit"
                        })
                    except (ValueError, TypeError):
                        pass

            if products:
                logger.info(f"Blinkit live scraper retrieved {len(products)} real products for '{query}'.")
                return products
        else:
            logger.warning(f"Blinkit layout search returned status {r.status_code}: {r.text[:200]}")
    except Exception as exc:
        logger.warning(f"Blinkit request error: {exc}")

    return products


async def fetch_blinkit(query: str, lat: str, lon: str) -> List[Dict[str, Any]]:
    return await asyncio.to_thread(fetch_blinkit_sync, query, lat, lon)


def fetch_zepto_sync(
    query: str,
    lat: str,
    lon: str,
    x_zepto_cookie: Optional[str] = None
) -> tuple[List[Dict[str, Any]], bool]:
    """
    Attempts to fetch live search results from Zepto search endpoints using curl_cffi with chrome124 impersonation.
    Attaches X-Zepto-Cookie (harvested by Flutter WebView) if provided.
    Returns (products, challenge_detected).
    """
    encoded_q = urllib.parse.quote(query)
    headers = {
        "Host": "www.zeptonow.com",
        "Referer": "https://www.zeptonow.com/",
        "Origin": "https://www.zeptonow.com",
        "User-Agent": "Mozilla/5.0 (Linux; Android 10; Mobile) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0.0.0 Mobile Safari/537.36",
        "Accept": "application/json, text/plain, */*",
        "Accept-Language": "en-US,en;q=0.9",
        "x-user-latitude": lat,
        "x-user-longitude": lon,
    }

    if x_zepto_cookie:
        headers["Cookie"] = x_zepto_cookie
        logger.info(f"Attached X-Zepto-Cookie ({len(x_zepto_cookie)} chars) to Zepto request.")

    urls_to_try = [
        f"https://www.zeptonow.com/api/v1/search/products?query={encoded_q}",
        f"https://www.zeptonow.com/search?query={encoded_q}"
    ]

    products = []
    challenge_detected = False
    for url in urls_to_try:
        try:
            r = requests.get(url, headers=headers, impersonate="chrome124", timeout=7)
            if r.status_code == 200:
                data = r.json()
                raw_items = (
                    data.get("products")
                    or data.get("items")
                    or (data.get("data") or {}).get("items")
                    or (data.get("data") or {}).get("products")
                    or []
                )

                # Also search inside widgets/layout if present
                if not raw_items and "layout" in data:
                    for widget in data.get("layout", []):
                        items = widget.get("data", {}).get("items", [])
                        if items:
                            raw_items.extend(items)

                for it in raw_items:
                    if not isinstance(it, dict):
                        continue
                    name = it.get("name") or it.get("product_name") or it.get("title") or ""
                    # Selling price (discounted_selling_price / 100 or mrp / 100)
                    raw_price = it.get("discounted_selling_price") or it.get("selling_price") or it.get("mrp") or 0.0
                    try:
                        price = float(raw_price) / 100.0 if float(raw_price) > 500 else float(raw_price)
                    except (ValueError, TypeError):
                        price = 0.0

                    unit = str(it.get("formatted_packsize") or it.get("packsize") or it.get("quantity") or "")
                    img = it.get("image_url") or it.get("image") or ""

                    if name and price > 0:
                        products.append({
                            "name": name,
                            "price": round(price, 2),
                            "unit": unit,
                            "image_url": img,
                            "store": "Zepto"
                        })
                if products:
                    logger.info(f"Zepto live scraper retrieved {len(products)} real products for '{query}'.")
                    return products, False
            elif r.status_code in [202, 403, 405]:
                logger.warning(f"Zepto endpoint '{url}' restricted by AWS WAF JavaScript challenge ({r.status_code}).")
                challenge_detected = True
            else:
                logger.warning(f"Zepto endpoint '{url}' returned status {r.status_code}.")
        except Exception as exc:
            logger.warning(f"Zepto request error for '{url}': {exc}")

    return products, challenge_detected


async def fetch_zepto(
    query: str,
    lat: str,
    lon: str,
    x_zepto_cookie: Optional[str] = None
) -> tuple[List[Dict[str, Any]], bool]:
    return await asyncio.to_thread(fetch_zepto_sync, query, lat, lon, x_zepto_cookie)


@app.get("/api/compare")
async def compare_prices(
    q: str = Query(..., description="Search query"),
    lat: str = Query(DEFAULT_LAT, description="User latitude"),
    lon: str = Query(DEFAULT_LON, description="User longitude"),
    x_zepto_cookie: Optional[str] = Header(None, alias="X-Zepto-Cookie")
):
    """
    Fires both live scrapers concurrently via asyncio.gather.
    Fuzzy-matches items using thefuzz.token_sort_ratio with threshold >= 60.
    Compares normalized unit prices.
    Accepts X-Zepto-Cookie from client WebView to bypass AWS WAF.
    """
    cleaned_q = q.strip()
    if not cleaned_q:
        return {"query": q, "results": [], "raw_blinkit_count": 0, "raw_zepto_count": 0, "zepto_blocked": False}

    # Execute both live scrapers concurrently
    blinkit_task = fetch_blinkit(cleaned_q, lat, lon)
    zepto_task = fetch_zepto(cleaned_q, lat, lon, x_zepto_cookie)
    blinkit_items, (zepto_items, zepto_challenged) = await asyncio.gather(blinkit_task, zepto_task)

    zepto_blocked = zepto_challenged and len(zepto_items) == 0

    paired_results = []
    used_zepto = set()

    # Pair items if both platforms returned results
    if blinkit_items and zepto_items:
        for b_item in blinkit_items:
            best_score = -1
            best_z_idx = None
            b_label = f"{b_item['name']} {b_item.get('unit', '')}".lower()

            for z_idx, z_item in enumerate(zepto_items):
                if z_idx in used_zepto:
                    continue
                z_label = f"{z_item['name']} {z_item.get('unit', '')}".lower()
                score = fuzz.token_sort_ratio(b_label, z_label)
                if score > best_score:
                    best_score = score
                    best_z_idx = z_idx

            if best_z_idx is not None and best_score >= 60:
                used_zepto.add(best_z_idx)
                z_item = zepto_items[best_z_idx]

                b_unit = calculate_unit_price(b_item["name"], b_item.get("unit", ""), b_item["price"])
                z_unit = calculate_unit_price(z_item["name"], z_item.get("unit", ""), z_item["price"])

                if b_unit["normalized_price"] < z_unit["normalized_price"]:
                    cheaper = "Blinkit"
                elif z_unit["normalized_price"] < b_unit["normalized_price"]:
                    cheaper = "Zepto"
                else:
                    cheaper = "Equal"

                paired_results.append({
                    "name": b_item["name"],
                    "packsize": b_item.get("unit") or z_item.get("unit", ""),
                    "unit": b_unit["unit"],
                    "match_score": best_score,
                    "cheaper_store": cheaper,
                    "blinkit": {
                        "name": b_item["name"],
                        "packsize": b_item.get("unit", ""),
                        "price": b_item["price"],
                        "normalized_price": b_unit["normalized_price"],
                        "unit": b_unit["unit"],
                        "formatted_unit_price": b_unit["formatted"],
                        "buy_url": f"https://blinkit.com/s/?q={urllib.parse.quote(b_item['name'])}"
                    },
                    "zepto": {
                        "name": z_item["name"],
                        "packsize": z_item.get("unit", ""),
                        "price": z_item["price"],
                        "normalized_price": z_unit["normalized_price"],
                        "unit": z_unit["unit"],
                        "formatted_unit_price": z_unit["formatted"],
                        "buy_url": f"https://www.zeptonow.com/search?query={urllib.parse.quote(z_item['name'])}"
                    }
                })
            else:
                # Unpaired Blinkit item
                b_unit = calculate_unit_price(b_item["name"], b_item.get("unit", ""), b_item["price"])
                paired_results.append({
                    "name": b_item["name"],
                    "packsize": b_item.get("unit", ""),
                    "unit": b_unit["unit"],
                    "match_score": 0,
                    "cheaper_store": "Blinkit",
                    "blinkit": {
                        "name": b_item["name"],
                        "packsize": b_item.get("unit", ""),
                        "price": b_item["price"],
                        "normalized_price": b_unit["normalized_price"],
                        "unit": b_unit["unit"],
                        "formatted_unit_price": b_unit["formatted"],
                        "buy_url": f"https://blinkit.com/s/?q={urllib.parse.quote(b_item['name'])}"
                    },
                    "zepto": None
                })
    elif blinkit_items:
        # Live items from Blinkit
        for b_item in blinkit_items:
            b_unit = calculate_unit_price(b_item["name"], b_item.get("unit", ""), b_item["price"])
            paired_results.append({
                "name": b_item["name"],
                "packsize": b_item.get("unit", ""),
                "unit": b_unit["unit"],
                "match_score": 0,
                "cheaper_store": "Blinkit",
                "blinkit": {
                    "name": b_item["name"],
                    "packsize": b_item.get("unit", ""),
                    "price": b_item["price"],
                    "normalized_price": b_unit["normalized_price"],
                    "unit": b_unit["unit"],
                    "formatted_unit_price": b_unit["formatted"],
                    "buy_url": f"https://blinkit.com/s/?q={urllib.parse.quote(b_item['name'])}"
                },
                "zepto": None
            })
    elif zepto_items:
        # Live items from Zepto
        for z_item in zepto_items:
            z_unit = calculate_unit_price(z_item["name"], z_item.get("unit", ""), z_item["price"])
            paired_results.append({
                "name": z_item["name"],
                "packsize": z_item.get("unit", ""),
                "unit": z_unit["unit"],
                "match_score": 0,
                "cheaper_store": "Zepto",
                "blinkit": None,
                "zepto": {
                    "name": z_item["name"],
                    "packsize": z_item.get("unit", ""),
                    "price": z_item["price"],
                    "normalized_price": z_unit["normalized_price"],
                    "unit": z_unit["unit"],
                    "formatted_unit_price": z_unit["formatted"],
                    "buy_url": f"https://www.zeptonow.com/search?query={urllib.parse.quote(z_item['name'])}"
                }
            })

    return {
        "query": cleaned_q,
        "results": paired_results,
        "raw_blinkit_count": len(blinkit_items),
        "raw_zepto_count": len(zepto_items),
        "zepto_blocked": zepto_blocked,
        "has_zepto_cookie": bool(x_zepto_cookie)
    }

class StoreOffer {
  final String name;
  final String packsize;
  final double price;
  final double normalizedPrice;
  final String unit;
  final String formattedUnitPrice;
  final String buyUrl;

  StoreOffer({
    required this.name,
    required this.packsize,
    required this.price,
    required this.normalizedPrice,
    required this.unit,
    required this.formattedUnitPrice,
    required this.buyUrl,
  });

  factory StoreOffer.fromJson(Map<String, dynamic> json) {
    return StoreOffer(
      name: json['name']?.toString() ?? '',
      packsize: json['packsize']?.toString() ?? '',
      price: (json['price'] as num?)?.toDouble() ?? 0.0,
      normalizedPrice: (json['normalized_price'] as num?)?.toDouble() ?? 0.0,
      unit: json['unit']?.toString() ?? '',
      formattedUnitPrice: json['formatted_unit_price']?.toString() ?? '',
      buyUrl: json['buy_url']?.toString() ?? '',
    );
  }
}

class ProductComparisonItem {
  final String name;
  final String packsize;
  final String unit;
  final int matchScore;
  final String cheaperStore;
  final StoreOffer? blinkit;
  final StoreOffer? zepto;

  ProductComparisonItem({
    required this.name,
    required this.packsize,
    required this.unit,
    required this.matchScore,
    required this.cheaperStore,
    this.blinkit,
    this.zepto,
  });

  factory ProductComparisonItem.fromJson(Map<String, dynamic> json) {
    return ProductComparisonItem(
      name: json['name']?.toString() ?? '',
      packsize: json['packsize']?.toString() ?? '',
      unit: json['unit']?.toString() ?? '',
      matchScore: (json['match_score'] as num?)?.toInt() ?? 0,
      cheaperStore: json['cheaper_store']?.toString() ?? '',
      blinkit: json['blinkit'] != null ? StoreOffer.fromJson(json['blinkit']) : null,
      zepto: json['zepto'] != null ? StoreOffer.fromJson(json['zepto']) : null,
    );
  }
}

            await _harvestCookies();
          },
        ),
      )
      ..loadRequest(Uri.parse('https://www.zeptonow.com'));
  }

  Future<void> _harvestCookies() async {
    try {
      final result = await _webViewController.runJavaScriptReturningResult('document.cookie');
      String cookieStr = result.toString().trim();
      if (cookieStr.startsWith('"') && cookieStr.endsWith('"') && cookieStr.length >= 2) {
        cookieStr = cookieStr.substring(1, cookieStr.length - 1);
      }
      cookieStr = cookieStr.replaceAll(r'\"', '"').replaceAll(r'\\', r'\');
      if (cookieStr.isNotEmpty && cookieStr != 'null') {
        if (mounted) {
          setState(() {
            _zeptoCookie = cookieStr;
            _hasHarvestedCookie = true;
          });
        }
        debugPrint('[WebView Harvester] Harvested cookie string (${cookieStr.length} chars)');
      }
    } catch (e) {
      debugPrint('[WebView Harvester] Error reading document.cookie: $e');
    }
  }

  Future<void> _initLocationAndFetch() async {

  Future<void> _fetchCurrentLocation() async {
    setState(() {
      _locationStatus = 'Locating...';
    });

    try {
      bool serviceEnabled = await Geolocator.isLocationServiceEnabled();
      if (!serviceEnabled) {
        setState(() {
          _locationStatus = 'Nagpur (Default GPS off)';
          _isGpsActive = false;
        });
        return;
      }

      LocationPermission permission = await Geolocator.checkPermission();
      if (permission == LocationPermission.denied) {
        permission = await Geolocator.requestPermission();
        if (permission == LocationPermission.denied) {
          setState(() {
            _locationStatus = 'Nagpur (Permission denied)';
            _isGpsActive = false;
          });
          return;
        }
      }

      if (permission == LocationPermission.deniedForever) {
        setState(() {
          _locationStatus = 'Nagpur (Permission denied)';
          _isGpsActive = false;
        });
        return;
      }

      Position position = await Geolocator.getCurrentPosition(
        desiredAccuracy: LocationAccuracy.high,
        timeLimit: const Duration(seconds: 8),
      );

      setState(() {
        _latitude = position.latitude;
        _longitude = position.longitude;
        _locationStatus = '${_latitude.toStringAsFixed(3)}, ${_longitude.toStringAsFixed(3)}';
        _isGpsActive = true;
      });
    } catch (e) {
      setState(() {
        _locationStatus = 'Nagpur (21.145, 79.088)';
        _isGpsActive = false;
      });
    }
  }

  Future<void> _searchGroceries(String query, {bool isRetry = false}) async {
    if (query.isEmpty) return;

    setState(() {
      _isLoading = true;
      _errorMessage = null;
    });

    final String encodedQuery = Uri.encodeComponent(query);
    final String url = 'http://$_serverHost:$_serverPort/api/compare?q=$encodedQuery&lat=$_latitude&lon=$_longitude';

    final headers = <String, String>{
      'Accept': 'application/json',
    };
    if (_zeptoCookie.isNotEmpty) {
      headers['X-Zepto-Cookie'] = _zeptoCookie;
    }

    try {
      final response = await http.get(
        Uri.parse(url),
        headers: headers,
      ).timeout(const Duration(seconds: 15));

      if (response.statusCode == 200) {
        final data = json.decode(response.body) as Map<String, dynamic>;

        // Self-healing: If Zepto was challenged / blocked, silently reload WebView and retry
        if (!isRetry && data['zepto_blocked'] == true) {
          debugPrint('[WebView Harvester] Zepto challenge detected. Silently reloading WebView and retrying...');
          await _reloadAndRetry(query);
          return;
        }

        final List<dynamic> rawList = data['results'] as List<dynamic>? ?? [];
        final parsedItems = rawList.map((e) => ProductComparisonItem.fromJson(e as Map<String, dynamic>)).toList();

        setState(() {
          _results = parsedItems;
          _rawBlinkitCount = (data['raw_blinkit_count'] as num?)?.toInt() ?? 0;
          _rawZeptoCount = (data['raw_zepto_count'] as num?)?.toInt() ?? 0;
          _isLoading = false;
        });
      } else if (response.statusCode == 403 && !isRetry) {
        debugPrint('[WebView Harvester] 403 received from backend. Silently reloading WebView and retrying...');
        await _reloadAndRetry(query);
      } else {
        setState(() {
          _errorMessage = 'Server error (${response.statusCode}): ${response.reasonPhrase}';
          _isLoading = false;
        });
      }
    } catch (e) {
      setState(() {
        _errorMessage = 'Could not reach backend at $url\n\nEnsure laptop and phone are on the same Wi-Fi and server.py is running on port 8000.';
        _isLoading = false;
      });
    }
  }

  Future<void> _reloadAndRetry(String query) async {
    try {
      await _webViewController.reload();
      await Future.delayed(const Duration(seconds: 3));
      await _harvestCookies();
      await _searchGroceries(query, isRetry: true);
    } catch (e) {
      debugPrint('[WebView Harvester] Error in reload and retry: $e');
    }
  }

          SnackBar(content: Text('Could not open link: $e')),
        );
      }
    }
  }

  final String serverHost;

    required this.onLoginSuccess,
  });

  @override
  State<_ZeptoLoginBottomSheet> createState() => _ZeptoLoginBottomSheetState();
}

class _ZeptoLoginBottomSheetState extends State<_ZeptoLoginBottomSheet> {
  int _stage = 1; // 1: Enter Phone, 2: Enter OTP
  final TextEditingController _phoneController = TextEditingController();
  final TextEditingController _otpController = TextEditingController();

  bool _isLoading = false;
  String? _errorMessage;
  String? _infoMessage;

  @override
  void dispose() {
    _phoneController.dispose();
    _otpController.dispose();
    super.dispose();
  }

  Future<void> _sendOtp() async {
    final phone = _phoneController.text.trim();
    final digits = phone.replaceAll(RegExp(r'\D'), '');
    final cleanPhone = digits.startsWith('91') && digits.length == 12 ? digits.substring(2) : digits;

    if (cleanPhone.length != 10) {
      setState(() {
        _errorMessage = 'Please enter a valid 10-digit mobile number';
      });
      return;
    }

    setState(() {
      _isLoading = true;
      _errorMessage = null;
      _infoMessage = null;
    });

    final url = 'http://${widget.serverHost}:${widget.serverPort}/api/auth/zepto/send-otp';
    try {
      final res = await http.post(
        Uri.parse(url),
        headers: {'Content-Type': 'application/json', 'Accept': 'application/json'},
        body: jsonEncode({'phone': cleanPhone}),
      ).timeout(const Duration(seconds: 12));

      final data = jsonDecode(res.body) as Map<String, dynamic>;
      if (res.statusCode == 200 && data['status'] == 'success') {
        setState(() {
          _stage = 2;
          _isLoading = false;
          _infoMessage = data['message'] ?? 'OTP sent to +91 $cleanPhone';
        });
      } else {
        setState(() {
          _isLoading = false;
          _errorMessage = data['message'] ?? 'Failed to send OTP (Status ${res.statusCode})';
        });
      }
    } catch (e) {
      setState(() {
        _isLoading = false;
        _errorMessage = 'Connection error: Could not reach backend at $url';
      });
    }
  }

  Future<void> _verifyOtp() async {
    final phone = _phoneController.text.trim();
    final digits = phone.replaceAll(RegExp(r'\D'), '');
    final cleanPhone = digits.startsWith('91') && digits.length == 12 ? digits.substring(2) : digits;
    final otp = _otpController.text.trim();

    if (otp.isEmpty || otp.length < 4) {
      setState(() {
        _errorMessage = 'Please enter the 4-digit OTP';
      });
      return;
    }

    setState(() {
      _isLoading = true;
      _errorMessage = null;
      _infoMessage = null;
    });

    final url = 'http://${widget.serverHost}:${widget.serverPort}/api/auth/zepto/verify-otp';
    try {
      final res = await http.post(
        Uri.parse(url),
        headers: {'Content-Type': 'application/json', 'Accept': 'application/json'},
        body: jsonEncode({'phone': cleanPhone, 'otp': otp}),
      ).timeout(const Duration(seconds: 12));

      final data = jsonDecode(res.body) as Map<String, dynamic>;
      if (res.statusCode == 200 && data['status'] == 'success') {
        setState(() {
          _isLoading = false;
        });
        if (mounted) {
          Navigator.pop(context);
          ScaffoldMessenger.of(context).showSnackBar(
            SnackBar(
              content: const Row(
                children: [
                  Icon(Icons.check_circle, color: Colors.white, size: 20),
                  SizedBox(width: 8),
                  Text('Zepto Linked! Showing live prices.', style: TextStyle(fontWeight: FontWeight.bold)),
                ],
              ),
              backgroundColor: const Color(0xFF059669),
              duration: const Duration(seconds: 3),
              behavior: SnackBarBehavior.floating,
              shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(10)),
            ),
          );
        }
        widget.onLoginSuccess();
      } else {
        setState(() {
          _isLoading = false;
          _errorMessage = data['message'] ?? 'Invalid OTP code';
        });
      }
    } catch (e) {
      setState(() {
        _isLoading = false;
        _errorMessage = 'Verification error: Could not reach backend at $url';
      });
    }
  }
